cmake_minimum_required(VERSION 3.20)
project(jogo_da_memoria CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(memory_core core/game.cpp)
target_include_directories(memory_core PUBLIC core)
if(NOT MSVC)
  target_compile_options(memory_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE memory_core)
add_test(NAME core_tests COMMAND core_tests)

# Jogo com janela (SDL2). Se o SDL2 não estiver instalado, só o núcleo e os testes são gerados.
find_package(SDL2 QUIET)
find_package(SDL2_image QUIET)
find_package(SDL2_ttf QUIET)
if(SDL2_FOUND AND SDL2_image_FOUND AND SDL2_ttf_FOUND)
  add_executable(memoria src/main.cpp)
  target_link_libraries(memoria PRIVATE memory_core SDL2::SDL2 SDL2_image::SDL2_image SDL2_ttf::SDL2_ttf)
  target_compile_definitions(memoria PRIVATE MEMORIA_SOURCE_ASSETS="${CMAKE_SOURCE_DIR}/assets")
  if(NOT MSVC)
    target_compile_options(memoria PRIVATE -Wall -Wextra)
  endif()
else()
  message(STATUS "SDL2/SDL2_image/SDL2_ttf não encontrados: o jogo (memoria) não será compilado")
endif()
